def card(date_ru, date_en, role_ru, role_en, company_ru, company_en, bullets, tags=None):
    lis = []
    for ru, en in bullets:
        lis.append(f'                            <li data-ru="{ru}" data-en="{en}">{ru}</li>')
    tags_html = ''
    if tags:
        chips = ''.join(f'<span class="timeline-tag">{x}</span>' for x in tags)
        tags_html = f'\n                        <div class="timeline-tags">{chips}</div>'
    if role_ru != role_en:
        role_html = f'<div class="timeline-role" data-ru="{role_ru}" data-en="{role_en}">{role_ru}</div>'
    else:
        role_html = f'<div class="timeline-role">{role_ru}</div>'
    return f'''            <div class="timeline-item">
                <div class="timeline-dot"></div>
                <div class="timeline-content spotlight" onclick="toggleDetails(this)">
                    <div class="timeline-date" data-ru="{date_ru}" data-en="{date_en}">{date_ru}</div>
                    {role_html}
                    <div class="timeline-company" data-ru="{company_ru}" data-en="{company_en}">{company_ru}</div>
                    <div class="expand-hint" data-ru="▸ Нажмите, чтобы раскрыть" data-en="▸ Click to expand">▸ Нажмите, чтобы раскрыть</div>
                    <div class="timeline-details">
                        <ul>
{chr(10).join(lis)}
                        </ul>{tags_html}
                    </div>
                </div>
            </div>'''
